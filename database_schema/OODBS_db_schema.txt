-- Organization Table
CREATE TABLE Organization (
    organization_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255),
    type VARCHAR(50),
    contact_email VARCHAR(255)
);

-- Users Table
CREATE TABLE Users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    organization_id INT,
    email VARCHAR(255),
    password_hash VARCHAR(255),
    full_name VARCHAR(255),
    role VARCHAR(50),
    FOREIGN KEY (organization_id) REFERENCES Organization(organization_id)
);

-- Pickup_Location Table
CREATE TABLE Pickup_Location (
    pickup_id INT AUTO_INCREMENT PRIMARY KEY,
    organization_id INT,
    type VARCHAR(50),
    name VARCHAR(255),
    latitude FLOAT,
    longitude FLOAT,
    FOREIGN KEY (organization_id) REFERENCES Organization(organization_id)
);

-- Organization_Locations Table
CREATE TABLE Organization_Locations (
    location_id INT AUTO_INCREMENT PRIMARY KEY,
    organization_id INT,
    name VARCHAR(255),
    latitude FLOAT,
    longitude FLOAT,
    FOREIGN KEY (organization_id) REFERENCES Organization(organization_id)
);

-- Bus Table
CREATE TABLE Bus (
    bus_id INT AUTO_INCREMENT PRIMARY KEY,
    organization_id INT,
    plate_number VARCHAR(50),
    driver_name VARCHAR(255),
    driver_phone_num INT,
    capacity INT,
    company VARCHAR(50),
    status VARCHAR(50),
    FOREIGN KEY (organization_id) REFERENCES Organization(organization_id)
);

-- Bus_Seats Table
CREATE TABLE Bus_Seats (
    seat_id INT AUTO_INCREMENT PRIMARY KEY,
    bus_id INT,
    seat_number VARCHAR(10),
    is_reserved BOOLEAN,
    FOREIGN KEY (bus_id) REFERENCES Bus(bus_id)
);

-- Bus_Services Table
CREATE TABLE Bus_Services (
    service_id INT AUTO_INCREMENT PRIMARY KEY,
    bus_id INT,
    pickup_id INT,
    service_date TIMESTAMP,
    isAmShift BOOLEAN,
    isPmShift BOOLEAN,
    FOREIGN KEY (bus_id) REFERENCES Bus(bus_id),
    FOREIGN KEY (pickup_id) REFERENCES Pickup_Location(pickup_id)
);

-- Schedule Table
CREATE TABLE Schedule (
    schedule_id INT AUTO_INCREMENT PRIMARY KEY,
    service_id INT,
    departure_time TIMESTAMP,
    arrival_time TIMESTAMP,
    FOREIGN KEY (service_id) REFERENCES Bus_Services(service_id)
);

-- Tier Table
CREATE TABLE Tier (
    tier_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(10)
);

-- Passenger_Requests Table
CREATE TABLE Passenger_Requests (
    request_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    bus_id INT,
    pickup_id INT,
    location_id INT,
    tier_id INT,
    passenger_count VARCHAR(50),
    request_status BOOLEAN,
    FOREIGN KEY (user_id) REFERENCES Users(user_id),
    FOREIGN KEY (bus_id) REFERENCES Bus(bus_id),
    FOREIGN KEY (pickup_id) REFERENCES Pickup_Location(pickup_id),
    FOREIGN KEY (location_id) REFERENCES Organization_Locations(location_id),
    FOREIGN KEY (tier_id) REFERENCES Tier(tier_id)
);

-- Seat_Assignments Table
CREATE TABLE Seat_Assignments (
    assignment_id INT AUTO_INCREMENT PRIMARY KEY,
    request_id INT,
    seat_id INT,
    FOREIGN KEY (request_id) REFERENCES Passenger_Requests(request_id),
    FOREIGN KEY (seat_id) REFERENCES Bus_Seats(seat_id)
);

-- Routes Table
CREATE TABLE Routes (
    route_id INT AUTO_INCREMENT PRIMARY KEY,
    schedule_id INT,
    request_id INT,
    tier_id INT,
    stop_order INT,
    eta TIMESTAMP,
    FOREIGN KEY (schedule_id) REFERENCES Schedule(schedule_id),
    FOREIGN KEY (request_id) REFERENCES Passenger_Requests(request_id),
    FOREIGN KEY (tier_id) REFERENCES Tier(tier_id)
);
